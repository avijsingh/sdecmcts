import torch
import tqdm

# ncbf = NeuralCBF(ckpt_path='outputs/cbf.ckpt')
ncbf = NeuralCBF(ckpt_path='../hw2/outputs/cbf.ckpt')

lo = torch.tensor([-3,-3,-3,-1,-1,-1,-1,-5,-5,-5,-5,-5,-5], dtype=torch.float32)
hi = torch.tensor([ 3, 3, 3, 1, 1, 1, 1, 5, 5, 5, 5, 5, 5], dtype=torch.float32)

total_volume = torch.prod(hi - lo).item()

batch_size = 1000
N = batch_size * 100000
safe_count = 0

# for i in tqdm.tqdm(range(N // batch_size)):
#     state = torch.rand(batch_size, 13) * (hi - lo).cpu() + lo.cpu()
#     vals = ncvf.values(state)
#     safe_count += (vals > 0).sum().item()

for j in tqdm.tqdm(range(N//batch_size)):
    state = torch.rand(batch_size, 13) * (hi - lo).cpu() + lo.cpu()
    vals = ncbf.values(state)
    safe_count += (vals > 0).sum().item()

frac_safe = safe_count / N
safe_volume = frac_safe * total_volume
print(f"Safe set fraction: {frac_safe:.4f}")
print(f"HJR safe set volume: {safe_volume:.4f}")
print(f"Total state space volume: {total_volume:.2f}")
