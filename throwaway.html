import torch
import torch.nn as nn
import tqdm

class VNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.input_linear = nn.Linear(3, 512)
        self.layer_0_linear = nn.Linear(512, 512)
        self.layer_1_linear = nn.Linear(512, 512)
        self.layer_2_linear = nn.Linear(512, 512)
        self.output_linear = nn.Linear(512, 1)

    def forward(self, x):
        x = torch.tanh(self.input_linear(x))
        x = torch.tanh(self.layer_0_linear(x))
        x = torch.tanh(self.layer_1_linear(x))
        x = torch.tanh(self.layer_2_linear(x))
        
        return self.output_linear(x).squeeze(-1)

ckpt = torch.load('../hw2/outputs/cbf.ckpt', map_location='cpu')
model = VNN()
state_dict = {k.replace('V_nn.', ''): v for k, v in ckpt['state_dict'].items()}
model.load_state_dict(state_dict)
model.eval()

lo = torch.tensor([-3,-3,-3,-1,-1,-1,-1,-5,-5,-5,-5,-5,-5], dtype=torch.float32)
hi = torch.tensor([ 3, 3, 3, 1, 1, 1, 1, 5, 5, 5, 5, 5, 5], dtype=torch.float32)
total_volume = torch.prod(hi - lo).item()

batch_size = 1000
N = batch_size * 10000
safe_count = 0

for i in tqdm.tqdm(range(N // batch_size)):
    state = torch.rand(batch_size, 13) * (hi - lo) + lo
    with torch.no_grad():
        vals = model(state[:, :3])
    safe_count += (vals > 0).sum().item()

frac_safe = safe_count / N
safe_volume = frac_safe * total_volume
print(f"CBF safe set fraction: {frac_safe:.4f}")
print(f"CBF safe set volume: {safe_volume:.4f}")
print(f"Total volume: {total_volume:.2f}")
