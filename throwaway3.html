import numpy as np
import jax.numpy as jnp
import hj_reachability as hj
from problem3_helper import save_values_gif as svg
from problem3_helper import plot_value_and_safe_set_boundary
from matplotlib import pyplot as plt
from scipy.interpolate import RegularGridInterpolator

m = 2.0
l = 1.0
g = 10.0
u_max = 3.0

class InvertedPendulum(hj.ControlAndDisturbanceAffineDynamics):
    control_space = hj.sets.Box(lo=jnp.array([-u_max]), hi=jnp.array([u_max]))
    disturbance_space = hj.sets.Box(lo=jnp.array([0.0]), hi=jnp.array([0.0]))

    def open_loop_dynamics(self, state, time):
        theta = state[0]
        theta_dot = state[1]
        return jnp.array([theta_dot, (g/l)*jnp.sin(theta)])

    def control_jacobian(self, state, time):
        return jnp.array([[0.0], [1.0/(m*l**2)]])
    
    def disturbance_jacobian(self, state, time):
        return jnp.array([[0.0], [0.0]])

grid = hj.Grid.from_lattice_parameters_and_boundary_conditions(domain=hj.sets.Box(lo=jnp.array([-np.pi, -10.0]), hi=jnp.array([np.pi, 10.0])), shape=(101, 101), periodic_dims=0)
theta_grid = grid.states[..., 0]
initial_values = 0.3 - np.abs(np.array(theta_grid))

solver_settings = hj.SolverSettings.with_accuracy("very_high", hamiltonian_postprocessor=hj.solver.backwards_reachable_tube)

times = np.linspace(0, -5, 101)
dynamics = InvertedPendulum(control_mode="max", disturbance_mode="min", control_space=hj.sets.Box(lo=jnp.array([-u_max]), hi=jnp.array([u_max])), disturbance_space=hj.sets.Box(lo=jnp.array([0.0]), hi=jnp.array([0.0])))
values = hj.solve(solver_settings, dynamics, grid, times, initial_values)

fig, axes = plt.subplots(3, 2, figsize=(20, 20))
vbar = 3

for i, t in enumerate((-5, -3, -2, -1, -0.5, 0)):
    ax = axes[i // 2][i % 2]
    ti = np.argwhere(np.isclose(times, t)).item()
    ax.set_title(f'$V(x, {times[ti]:3.2f})$')
    ax.set_xlabel('$\\theta$ (rad)')
    ax.set_ylabel('$\\dot{\\theta}$ (rad/s)')

    value_plot = ax.pcolormesh(
        grid.coordinate_vectors[0],
        grid.coordinate_vectors[1],
        values[ti].T,
        cmap='coolwarm',
        vmin=-vbar,
        vmax=+vbar
    )
    plt.colorbar(value_plot, ax=ax)

    value_contour = ax.contour(
        grid.coordinate_vectors[0],
        grid.coordinate_vectors[1],
        values[ti].T,
        levels=[0],
        colors='k'
    )
    ax.axvline(-0.3, color='purple', linestyle='--', linewidth=1.5)
    ax.axvline(0.3,  color='purple', linestyle='--', linewidth=1.5)

conv_val = values[-1]
ss_vol = np.prod(grid.domain.hi - grid.domain.lo)
safe_ratio = np.mean(conv_val > 0)
safe_vol = ss_vol * safe_ratio
print(f"State Space Volume: {ss_vol:.2f}")
print(f"Safe Set Volume: {safe_vol:.2f}")
print(f"Safe Set Volume (%): {safe_ratio * 100:.2f}")

a = 0.1
b = 0.2
ell_vol = a * b * np.pi
print(f"HW1 Volume: {ell_vol:.2f}")

theta_vals = np.array(grid.coordinate_vectors[0])
thetadot_vals = np.array(grid.coordinate_vectors[1])
V_converged = np.array(values[-1])

V_interp = RegularGridInterpolator(
    (theta_vals, thetadot_vals), V_converged,
    method='linear', bounds_error=False, fill_value=None
)

def get_gradient(state):
    eps = 1e-4
    dV_dtheta    = (V_interp([state + np.array([eps, 0])]) -
                    V_interp([state - np.array([eps, 0])])) / (2 * eps)
    dV_dthetadot = (V_interp([state + np.array([0, eps])]) -
                    V_interp([state - np.array([0, eps])])) / (2 * eps)
    return float(dV_dtheta[0]), float(dV_dthetadot[0])

grad_values = grid.grad_values(V_converged)

grad_interp_theta = RegularGridInterpolator((theta_vals, thetadot_vals), np.array(grad_values[..., 0]), method='linear', bounds_error=False, fill_value=0.0)
grad_interp_thetadot = RegularGridInterpolator((theta_vals, thetadot_vals), np.array(grad_values[..., 1]),method='linear', bounds_error=False, fill_value=0.0)

def optimal_control(state):
    dV_dthetadot = float(grad_interp_thetadot([state])[0])
    if dV_dthetadot >= 0:
        return u_max
    else:
        return -u_max

def pendulum_dynamics(state, u):
    theta, theta_dot = state
    return np.array([theta_dot, (g / l) * np.sin(theta) + u / (m * l**2)])

dt = 0.01
T = 1.0
t_sim = np.arange(0, T + dt, dt)

initial_states = [np.array([-0.1, +0.4]), np.array([-0.1, -0.3])]
colors = ['tab:blue', 'tab:orange']
labels = ['$x_0 = (-0.1, +0.4)$', '$x_0 = (-0.1, -0.3)$']

trajectories = []
controls = []

for x0 in initial_states:
    xs = [x0.copy()]
    us = []
    x  = x0.copy()
    for _ in t_sim[:-1]:
        u = optimal_control(x)
        us.append(u)
        x = x + dt * pendulum_dynamics(x, u)
        xs.append(x.copy())
    trajectories.append(np.array(xs))
    controls.append(np.array(us))

fig, ax = plt.subplots(figsize=(8, 6))
plot_value_and_safe_set_boundary(V_converged, grid, ax)
for traj, c, lbl in zip(trajectories, colors, labels):
    ax.plot(traj[:, 0], traj[:, 1], color=c, linewidth=2, label=lbl)
    ax.plot(traj[0, 0], traj[0, 1], 'o', color=c, markersize=8)
ax.set_xlim(-0.5, 0.5)
ax.set_ylim(-1.0,  1.0)
ax.set_xlabel('$\\theta$ (rad)')
ax.set_ylabel('$\\dot{\\theta}$ (rad/s)')
ax.set_title('Trajectories under optimal safety controller')
ax.legend()
plt.tight_layout()
plt.savefig('trajectories.png', dpi=150)

fig, ax = plt.subplots(figsize=(8, 4))
for us, c, lbl in zip(controls, colors, labels):
    ax.plot(t_sim[:-1], us, color=c, linewidth=2, label=lbl)
ax.axhline( u_max, color='k', linestyle='--', linewidth=1, label='$u_{max}$')
ax.axhline(-u_max, color='k', linestyle='--', linewidth=1, label='$-u_{max}$')
ax.set_xlabel('Time (s)')
ax.set_ylabel('$u$ (N·m)')
ax.set_title('Control inputs vs time')
ax.legend()
plt.tight_layout()
plt.savefig('control_inputs.png', dpi=150)
plt.show()
